63: class PiecewiseLinear(object):
64:     """
65:     Piecewise linear function, from float to float, specified as nonempty list of (x,y) pairs with
66:     the x values in order.  x values <[initial x] or >[final x] are map to [initial y], [final y]
67:     respectively.
68:     """
69: 
70:     def __init__(self, *args):
71:         assert len(args) >= 1, len(args)
72:         if len(args) == 1 and isinstance(args[0], PiecewiseLinear):
73:             self.pairs = list(args[0].pairs)
74:         else:
75:             self.pairs = [(float(x), float(y)) for x, y in args]
76:         for x, y in self.pairs:
77:             assert isinstance(x, (float, int)), type(x)
78:             assert isinstance(y, (float, int)), type(y)
79: 
80:         for i in range(len(self.pairs) - 1):
81:             assert self.pairs[i + 1][0] > self.pairs[i][0], (
82:                 i,
83:                 self.pairs[i],
84:                 self.pairs[i + 1],
85:             )
86: 
87:     def __str__(self):
88:         # e.g. 'PiecewiseLinear((0., 10.), (100., 0.))'
89:         return f"PiecewiseLinear({str(self.pairs)[1:-1]})"
90: 
91:     def __call__(self, x):
92:         if x <= self.pairs[0][0]:
93:             return self.pairs[0][1]
94:         elif x >= self.pairs[-1][0]:
95:             return self.pairs[-1][1]
96:         else:
97:             cur_x, cur_y = self.pairs[0]
98:             for i in range(1, len(self.pairs)):
99:                 next_x, next_y = self.pairs[i]
100:                 if x >= cur_x and x <= next_x:
101:                     return cur_y + (next_y - cur_y) * (x - cur_x) / (next_x - cur_x)
102:                 cur_x, cur_y = next_x, next_y
103:             assert False
104: 
105:     def __mul__(self, alpha):
106:         return PiecewiseLinear(*[(x, y * alpha) for x, y in self.pairs])
107: 
108:     def __add__(self, x):
109:         if isinstance(x, (float, int)):
110:             return PiecewiseLinear(*[(p[0], p[1] + x) for p in self.pairs])
111:         s, x = self.get_common_basis(x)
112:         return PiecewiseLinear(
113:             *[(sp[0], sp[1] + xp[1]) for sp, xp in zip(s.pairs, x.pairs)]
114:         )
115: 
116:     def max(self, x):
117:         if isinstance(x, (float, int)):
118:             x = PiecewiseLinear((0, x))
119:         s, x = self.get_common_basis(x, include_crossings=True)
120:         return PiecewiseLinear(
121:             *[(sp[0], max(sp[1], xp[1])) for sp, xp in zip(s.pairs, x.pairs)]
122:         )
123: 
124:     def min(self, x):
125:         if isinstance(x, float) or isinstance(x, int):
126:             x = PiecewiseLinear((0, x))
127:         s, x = self.get_common_basis(x, include_crossings=True)
128:         return PiecewiseLinear(
129:             *[(sp[0], min(sp[1], xp[1])) for sp, xp in zip(s.pairs, x.pairs)]
130:         )
131: 
132:     def __eq__(self, other):
133:         return self.pairs == other.pairs
134: 
135:     def get_common_basis(self, p: "PiecewiseLinear", include_crossings: bool = False):
136:         """
137:         Returns (self_mod, p_mod) which are equivalent piecewise linear
138:         functions to self and p, but with the same x values.
139: 
140:           p: the other piecewise linear function
141:           include_crossings: if true, include in the x values positions
142:               where the functions indicate by this and p cross.
143:         """
144:         assert isinstance(p, PiecewiseLinear), type(p)
145: 
146:         # get sorted x-values without repetition.
147:         x_vals = sorted(set([x for x, _ in self.pairs] + [x for x, _ in p.pairs]))
148:         y_vals1 = [self(x) for x in x_vals]
149:         y_vals2 = [p(x) for x in x_vals]
150: 
151:         if include_crossings:
152:             extra_x_vals = []
153:             for i in range(len(x_vals) - 1):
154:                 if (y_vals1[i] > y_vals2[i]) != (y_vals1[i + 1] > y_vals2[i + 1]):
155:                     # if the two lines in this subsegment potentially cross each other..
156:                     diff_cur = abs(y_vals1[i] - y_vals2[i])
157:                     diff_next = abs(y_vals1[i + 1] - y_vals2[i + 1])
158:                     # `pos`, between 0 and 1, gives the relative x position,
159:                     # with 0 being x_vals[i] and 1 being x_vals[i+1].
160:                     pos = diff_cur / (diff_cur + diff_next)
161:                     extra_x_val = x_vals[i] + pos * (x_vals[i + 1] - x_vals[i])
162:                     extra_x_vals.append(extra_x_val)
163:             if len(extra_x_vals) > 0:
164:                 x_vals = sorted(set(x_vals + extra_x_vals))
165: 
166:             y_vals1 = [self(x) for x in x_vals]
167:             y_vals2 = [p(x) for x in x_vals]
168: 
169:         return (
170:             PiecewiseLinear(*zip(x_vals, y_vals1)),
171:             PiecewiseLinear(*zip(x_vals, y_vals2)),
172:         )

175: class ScheduledFloat(torch.nn.Module):
176:     """
177:     This object is a torch.nn.Module only because we want it to show up in [top_level module].modules();
178:     it does not have a working forward() function.  You are supposed to cast it to float, as
179:     in, float(parent_module.whatever), and use it as something like a dropout prob.
180: 
181:     It is a floating point value whose value changes depending on the batch count of the
182:     training loop.  It is a piecewise linear function where you specify the (x,y) pairs
183:     in sorted order on x; x corresponds to the batch index.  For batch-index values before the
184:     first x or after the last x, we just use the first or last y value.
185: 
186:     Example:
187:        self.dropout = ScheduledFloat((0.0, 0.2), (4000.0, 0.0), default=0.0)
188: 
189:     `default` is used when self.batch_count is not set or not in training mode or in
190:      torch.jit scripting mode.
191:     """
192: 
193:     def __init__(self, *args, default: float = 0.0):
194:         super().__init__()
195:         # self.batch_count and self.name will be written to in the training loop.
196:         self.batch_count = None
197:         self.name = None
198:         self.default = default
199:         self.schedule = PiecewiseLinear(*args)
200: 
201:     def extra_repr(self) -> str:
202:         return (
203:             f"batch_count={self.batch_count}, schedule={str(self.schedule.pairs[1:-1])}"
204:         )
205: 
206:     def __float__(self):
207:         batch_count = self.batch_count
208:         if (
209:             batch_count is None
210:             or not self.training
211:             or torch.jit.is_scripting()
212:             or torch.jit.is_tracing()
213:         ):
214:             return float(self.default)
215:         else:
216:             ans = self.schedule(self.batch_count)
217:             if random.random() < 0.0002:
218:                 logging.info(
219:                     f"ScheduledFloat: name={self.name}, batch_count={self.batch_count}, ans={ans}"
220:                 )
221:             return ans
222: 
223:     def __add__(self, x):
224:         if isinstance(x, float) or isinstance(x, int):
225:             return ScheduledFloat(self.schedule + x, default=self.default)
226:         else:
227:             return ScheduledFloat(
228:                 self.schedule + x.schedule, default=self.default + x.default
229:             )
230: 
231:     def max(self, x):
232:         if isinstance(x, float) or isinstance(x, int):
233:             return ScheduledFloat(self.schedule.max(x), default=self.default)
234:         else:
235:             return ScheduledFloat(
236:                 self.schedule.max(x.schedule), default=max(self.default, x.default)
237:             )

243: def random_cast_to_half(x: Tensor, min_abs: float = 5.0e-06) -> Tensor:
244:     """
245:     A randomized way of casting a floating point value to half precision.
246:     """
247:     if x.dtype == torch.float16:
248:         return x
249:     x_abs = x.abs()
250:     is_too_small = x_abs < min_abs
251:     # for elements where is_too_small is true, random_val will contain +-min_abs with
252:     # probability (x.abs() / min_abs), and 0.0 otherwise.  [so this preserves expectations,
253:     # for those elements].
254:     random_val = min_abs * x.sign() * (torch.rand_like(x) * min_abs < x_abs)
255:     return torch.where(is_too_small, random_val, x).to(torch.float16)

258: class CutoffEstimator:
259:     """
260:     Estimates cutoffs of an arbitrary numerical quantity such that a specified
261:     proportion of items will be above the cutoff on average.
262: 
263:       p is the proportion of items that should be above the cutoff.
264:     """
265: 
266:     def __init__(self, p: float):
267:         self.p = p
268:         # total count of items
269:         self.count = 0
270:         # total count of items that were above the cutoff
271:         self.count_above = 0
272:         # initial cutoff value
273:         self.cutoff = 0
274: 
275:     def __call__(self, x: float) -> bool:
276:         """
277:         Returns true if x is above the cutoff.
278:         """
279:         ans = x > self.cutoff
280:         self.count += 1
281:         if ans:
282:             self.count_above += 1
283:         cur_p = self.count_above / self.count
284:         delta_p = cur_p - self.p
285:         if (delta_p > 0) == ans:
286:             q = abs(delta_p)
287:             self.cutoff = x * q + self.cutoff * (1 - q)
288:         return ans

328: class MaxEigLimiterFunction(torch.autograd.Function):
329:     @staticmethod
330:     def forward(
331:         ctx,
332:         x: Tensor,
333:         coeffs: Tensor,
334:         direction: Tensor,
335:         channel_dim: int,
336:         grad_scale: float,
337:     ) -> Tensor:
338:         ctx.channel_dim = channel_dim
339:         ctx.grad_scale = grad_scale
340:         ctx.save_for_backward(x.detach(), coeffs.detach(), direction.detach())
341:         return x
342: 
343:     @staticmethod
344:     def backward(ctx, x_grad, *args):
345:         with torch.enable_grad():
346:             (x_orig, coeffs, new_direction) = ctx.saved_tensors
347:             x_orig.requires_grad = True
348:             num_channels = x_orig.shape[ctx.channel_dim]
349:             x = x_orig.transpose(ctx.channel_dim, -1).reshape(-1, num_channels)
350:             new_direction.requires_grad = False
351:             x = x - x.mean(dim=0)
352:             x_var = (x**2).mean()
353:             x_residual = x - coeffs * new_direction
354:             x_residual_var = (x_residual**2).mean()
355:             # `variance_proportion` is the proportion of the variance accounted for
356:             # by the top eigen-direction.  This is to be minimized.
357:             variance_proportion = (x_var - x_residual_var) / (x_var + 1.0e-20)
358:             variance_proportion.backward()
359:         x_orig_grad = x_orig.grad
360:         x_extra_grad = (
361:             x_orig.grad
362:             * ctx.grad_scale
363:             * x_grad.norm()
364:             / (x_orig_grad.norm() + 1.0e-20)
365:         )
366:         return x_grad + x_extra_grad.detach(), None, None, None, None

369: class BiasNormFunction(torch.autograd.Function):
370:     # This computes:
371:     #   scales = (torch.mean((x - bias) ** 2, keepdim=True)) ** -0.5 * log_scale.exp()
372:     #   return x * scales
373:     # (after unsqueezing the bias), but it does it in a memory-efficient way so that
374:     # it can just store the returned value (chances are, this will also be needed for
375:     # some other reason, related to the next operation, so we can save memory).
376:     @staticmethod
377:     def forward(
378:         ctx,
379:         x: Tensor,
380:         bias: Tensor,
381:         log_scale: Tensor,
382:         channel_dim: int,
383:         store_output_for_backprop: bool,
384:     ) -> Tensor:
385:         assert bias.ndim == 1
386:         if channel_dim < 0:
387:             channel_dim = channel_dim + x.ndim
388:         ctx.store_output_for_backprop = store_output_for_backprop
389:         ctx.channel_dim = channel_dim
390:         for _ in range(channel_dim + 1, x.ndim):
391:             bias = bias.unsqueeze(-1)
392:         scales = (
393:             torch.mean((x - bias) ** 2, dim=channel_dim, keepdim=True) ** -0.5
394:         ) * log_scale.exp()
395:         ans = x * scales
396:         ctx.save_for_backward(
397:             ans.detach() if store_output_for_backprop else x,
398:             scales.detach(),
399:             bias.detach(),
400:             log_scale.detach(),
401:         )
402:         return ans
403: 
404:     @staticmethod
405:     def backward(ctx, ans_grad: Tensor) -> Tensor:
406:         ans_or_x, scales, bias, log_scale = ctx.saved_tensors
407:         if ctx.store_output_for_backprop:
408:             x = ans_or_x / scales
409:         else:
410:             x = ans_or_x
411:         x = x.detach()
412:         x.requires_grad = True
413:         bias.requires_grad = True
414:         log_scale.requires_grad = True
415:         with torch.enable_grad():
416:             # recompute scales from x, bias and log_scale.
417:             scales = (
418:                 torch.mean((x - bias) ** 2, dim=ctx.channel_dim, keepdim=True) ** -0.5
419:             ) * log_scale.exp()
420:             ans = x * scales
421:             ans.backward(gradient=ans_grad)
422:         return x.grad, bias.grad.flatten(), log_scale.grad, None, None

425: class BiasNorm(torch.nn.Module):
426:     """
427:     This is intended to be a simpler, and hopefully cheaper, replacement for
428:     LayerNorm.  The observation this is based on, is that Transformer-type
429:     networks, especially with pre-norm, sometimes seem to set one of the
430:     feature dimensions to a large constant value (e.g. 50), which "defeats"
431:     the LayerNorm because the output magnitude is then not strongly dependent
432:     on the other (useful) features.  Presumably the weight and bias of the
433:     LayerNorm are required to allow it to do this.
434: 
435:     Instead, we give the BiasNorm a trainable bias that it can use when
436:     computing the scale for normalization.  We also give it a (scalar)
437:     trainable scale on the output.
438: 
439: 
440:     Args:
441:        num_channels: the number of channels, e.g. 512.
442:        channel_dim: the axis/dimension corresponding to the channel,
443:          interpreted as an offset from the input's ndim if negative.
444:          This is NOT the num_channels; it should typically be one of
445:          {-2, -1, 0, 1, 2, 3}.
446:       log_scale: the initial log-scale that we multiply the output by; this
447:          is learnable.
448:       log_scale_min: FloatLike, minimum allowed value of log_scale
449:       log_scale_max: FloatLike, maximum allowed value of log_scale
450:       store_output_for_backprop: only possibly affects memory use; recommend
451:          to set to True if you think the output of this module is more likely
452:          than the input of this module to be required to be stored for the
453:          backprop.
454:     """
455: 
456:     def __init__(
457:         self,
458:         num_channels: int,
459:         channel_dim: int = -1,  # CAUTION: see documentation.
460:         log_scale: float = 1.0,
461:         log_scale_min: float = -1.5,
462:         log_scale_max: float = 1.5,
463:         store_output_for_backprop: bool = False,
464:     ) -> None:
465:         super(BiasNorm, self).__init__()
466:         self.num_channels = num_channels
467:         self.channel_dim = channel_dim
468:         self.log_scale = nn.Parameter(torch.tensor(log_scale))
469:         self.bias = nn.Parameter(torch.empty(num_channels).normal_(mean=0, std=1e-4))
470: 
471:         self.log_scale_min = log_scale_min
472:         self.log_scale_max = log_scale_max
473: 
474:         self.store_output_for_backprop = store_output_for_backprop
475: 
476:     def forward(self, x: Tensor) -> Tensor:
477:         assert x.shape[self.channel_dim] == self.num_channels
478: 
479:         if torch.jit.is_scripting() or torch.jit.is_tracing():
480:             channel_dim = self.channel_dim
481:             if channel_dim < 0:
482:                 channel_dim += x.ndim
483:             bias = self.bias
484:             for _ in range(channel_dim + 1, x.ndim):
485:                 bias = bias.unsqueeze(-1)
486:             scales = (
487:                 torch.mean((x - bias) ** 2, dim=channel_dim, keepdim=True) ** -0.5
488:             ) * self.log_scale.exp()
489:             return x * scales
490: 
491:         log_scale = limit_param_value(
492:             self.log_scale,
493:             min=float(self.log_scale_min),
494:             max=float(self.log_scale_max),
495:             training=self.training,
496:         )
497: 
498:         return BiasNormFunction.apply(
499:             x, self.bias, log_scale, self.channel_dim, self.store_output_for_backprop
500:         )

503: def ScaledLinear(*args, initial_scale: float = 1.0, **kwargs) -> nn.Linear:
504:     """
505:     Behaves like a constructor of a modified version of nn.Linear
506:     that gives an easy way to set the default initial parameter scale.
507: 
508:     Args:
509:         Accepts the standard args and kwargs that nn.Linear accepts
510:         e.g. in_features, out_features, bias=False.
511: 
512:         initial_scale: you can override this if you want to increase
513:            or decrease the initial magnitude of the module's output
514:            (affects the initialization of weight_scale and bias_scale).
515:            Another option, if you want to do something like this, is
516:            to re-initialize the parameters.
517:     """
518:     ans = nn.Linear(*args, **kwargs)
519:     with torch.no_grad():
520:         ans.weight[:] *= initial_scale
521:         if ans.bias is not None:
522:             torch.nn.init.uniform_(ans.bias, -0.1 * initial_scale, 0.1 * initial_scale)
523:     return ans

573: class ChunkCausalDepthwiseConv1d(torch.nn.Module):
574:     """
575:     Behaves like a depthwise 1d convolution, except that it is causal in
576:     a chunkwise way, as if we had a block-triangular attention mask.
577:     The chunk size is provided at test time (it should probably be
578:     kept in sync with the attention mask).

580:     This has a little more than twice the parameters of a conventional
581:     depthwise conv1d module: we implement it by having one
582:     depthwise convolution, of half the width, that is causal (via
583:     right-padding); and one depthwise convolution that is applied only
584:     within chunks, that we multiply by a scaling factor which depends
585:     on the position within the chunk.
586: 
587:     Args:
588:         Accepts the standard args and kwargs that nn.Linear accepts
589:         e.g. in_features, out_features, bias=False.

592:            or decrease the initial magnitude of the module's output
593:            (affects the initialization of weight_scale and bias_scale).
594:            Another option, if you want to do something like this, is
595:            to re-initialize the parameters.

598:     def __init__(

600:         channels: int,
601:         kernel_size: int,
602:         initial_scale: float = 1.0,
603:         bias: bool = True,

629:         # second row is correction factors added to the scale near the right edge of the chunk,
630:         # both of these are added to a default scale of 1.0.
631:         self.chunkwise_conv_scale = nn.Parameter(torch.zeros(2, channels, kernel_size))
632:         self.kernel_size = kernel_size

636:             self.chunkwise_conv.weight[:] *= initial_scale

642:     def forward(self, x: Tensor, chunk_size: int = -1) -> Tensor:
643:         """Forward function.
644: 
645:         Args:
646:                x: a Tensor of shape (batch_size, channels, seq_len)
647:         chunk_size: the chunk size, in frames; does not have to divide seq_len exactly.
648:         """
649:         (batch_size, num_channels, seq_len) = x.shape
650: 
651:         # half_kernel_size = self.kernel_size + 1 // 2
652:         # left_pad is half_kernel_size - 1 where half_kernel_size is the size used
653:         # in the causal conv.  It's the amount by which we must pad on the left,
654:         # to make the convolution causal.
655:         left_pad = self.kernel_size // 2
656: 
657:         if chunk_size < 0 or chunk_size > seq_len:
658:             chunk_size = seq_len
659:         right_pad = -seq_len % chunk_size
660: 
661:         x = torch.nn.functional.pad(x, (left_pad, right_pad))
662: 
663:         x_causal = self.causal_conv(x[..., : left_pad + seq_len])
664:         assert x_causal.shape == (batch_size, num_channels, seq_len)
665: 
666:         x_chunk = x[..., left_pad:]
667:         num_chunks = x_chunk.shape[2] // chunk_size
668:         x_chunk = x_chunk.reshape(batch_size, num_channels, num_chunks, chunk_size)
669:         x_chunk = x_chunk.permute(0, 2, 1, 3).reshape(
670:             batch_size * num_chunks, num_channels, chunk_size
671:         )
672:         x_chunk = self.chunkwise_conv(x_chunk)  # does not change shape
673: 
674:         chunk_scale = self._get_chunk_scale(chunk_size)
675: 
676:         x_chunk = x_chunk * chunk_scale

678:             batch_size, num_chunks, num_channels, chunk_size
679:         ).permute(0, 2, 1, 3)
680:         x_chunk = x_chunk.reshape(batch_size, num_channels, num_chunks * chunk_size)[
681:             ..., :seq_len
682:         ]
683: 
684:         return x_chunk + x_causal

686:     def _get_chunk_scale(self, chunk_size: int):
687:         """Returns tensor of shape (num_channels, chunk_size) that will be used to
688:         scale the output of self.chunkwise_conv."""
689:         left_edge = self.chunkwise_conv_scale[0]
690:         right_edge = self.chunkwise_conv_scale[1]
691:         if chunk_size < self.kernel_size:
692:             left_edge = left_edge[:, :chunk_size]
693:             right_edge = right_edge[:, -chunk_size:]
694:         else:
695:             t = chunk_size - self.kernel_size
696:             channels = left_edge.shape[0]
697:             pad = torch.zeros(
698:                 channels, t, device=left_edge.device, dtype=left_edge.dtype
699:             )
700:             left_edge = torch.cat((left_edge, pad), dim=-1)
701:             right_edge = torch.cat((pad, right_edge), dim=-1)
702:         return 1.0 + (left_edge + right_edge)

812: class Balancer(torch.nn.Module):
813:     """
814:     Modifies the backpropped derivatives of a function to try to encourage, for
815:     each channel, that it is positive at least a proportion `threshold` of the
816:     time.  It does this by multiplying negative derivative values by up to
817:     (1+max_factor), and positive derivative values by up to (1-max_factor),
818:     interpolated from 1 at the threshold to those extremal values when none
819:     of the inputs are positive.
820: 
821:     Args:
822:            num_channels: the number of channels
823:            channel_dim: the dimension/axis corresponding to the channel, e.g.
824:                -1, 0, 1, 2; will be interpreted as an offset from x.ndim if negative.
825:            min_positive: the minimum, per channel, of the proportion of the time
826:                that (x > 0), below which we start to modify the derivatives.
827:            max_positive: the maximum, per channel, of the proportion of the time
828:                that (x > 0), above which we start to modify the derivatives.
829:            scale_gain_factor: determines the 'gain' with which we increase the
830:               change in gradient once the constraints on min_abs and max_abs
831:               are violated.
832:            min_abs:  the minimum average-absolute-value difference from the mean
833:                value per channel, which we allow, before we start to modify
834:                the derivatives to prevent this.
835:            max_abs:  the maximum average-absolute-value difference from the mean
836:                value per channel, which we allow, before we start to modify
837:                the derivatives to prevent this.
838:          prob: determines the minimum probability with which we modify the
839:              gradients for the {min,max}_positive and {min,max}_abs constraints,
840:              on each forward().  This is done randomly to prevent all layers
841:              from doing it at the same time.
842:     """
843: 
844:     def __init__(
845:         self,
846:         num_channels: int,
847:         channel_dim: int,
848:         min_positive: FloatLike = 0.05,
849:         max_positive: FloatLike = 0.95,

859:         self.prob = prob
860:         # 5% of the time we will return and do nothing because memory usage is
861:         # too high.
862:         self.mem_cutoff = CutoffEstimator(0.05)

865:         self.num_channels = num_channels

874:         if (
875:             torch.jit.is_scripting()
876:             or not x.requires_grad
877:             or (x.is_cuda and self.mem_cutoff(torch.cuda.memory_allocated()))
878:         ):
879:             return _no_op(x)
880: 
881:         prob = float(self.prob)
882:         if random.random() < prob:
883:             # The following inner-functions convert from the way we historically specified
884:             # these limitations, as limits on the absolute value and the proportion of positive
885:             # values, to limits on the RMS value and the (mean / stddev).
886:             def _abs_to_rms(x):
887:                 # for normally distributed data, if the expected absolute value is x, the
888:                 # expected rms value will be sqrt(pi/2) * x.
889:                 return 1.25331413732 * x
890: 
891:             def _proportion_positive_to_mean(x):

893:                     eps = 1.0e-10

895:                     # we'll just end up returning a fairly large value.
896:                     return (math.log(1 + x + eps) - math.log(1 - x + eps)) / 2.0
897: 
898:                 def _approx_inverse_erf(x):
899:                     # 1 / (sqrt(pi) * ln(2)),
900:                     # see https://math.stackexchange.com/questions/321569/approximating-the-error-function-erf-by-analytical-functions
901:                     # this approximation is extremely crude and gets progressively worse for
902:                     # x very close to -1 or +1, but we mostly care about the "middle" region
903:                     # e.g. _approx_inverse_erf(0.05) = 0.0407316414078772,
904:                     # and math.erf(0.0407316414078772) = 0.045935330944660666,
905:                     # which is pretty close to 0.05.
906:                     return 0.8139535143 * _atanh(x)
907: 
908:                 # first convert x from the range 0..1 to the range -1..1 which the error
909:                 # function returns
910:                 x = -1 + (2 * x)
911:                 return _approx_inverse_erf(x)

914:             max_mean = _proportion_positive_to_mean(float(self.max_positive))
915:             min_rms = _abs_to_rms(float(self.min_abs))
916:             max_rms = _abs_to_rms(float(self.max_abs))
917:             grad_scale = float(self.grad_scale)

919:             assert x.shape[self.channel_dim] == self.num_channels

928: def penalize_abs_values_gt(
929:     x: Tensor, limit: float, penalty: float, name: str = None
930: ) -> Tensor:
931:     """
932:     Returns x unmodified, but in backprop will put a penalty for the excess of
933:     the absolute values of elements of x over the limit "limit".  E.g. if
934:     limit == 10.0, then if x has any values over 10 it will get a penalty.
935: 
936:     Caution: the value of this penalty will be affected by grad scaling used
937:     in automatic mixed precision training.  For this reasons we use this,
938:     it shouldn't really matter, or may even be helpful; we just use this
939:     to disallow really implausible values of scores to be given to softmax.
940: 
941:     The name is for randomly printed debug info.
942:     """
943:     x_sign = x.sign()
944:     over_limit = (x.abs() - limit) > 0
945:     # The following is a memory efficient way to penalize the absolute values of
946:     # x that's over the limit.  (The memory efficiency comes when you think
947:     # about which items torch needs to cache for the autograd, and which ones it
948:     # can throw away).  The numerical value of aux_loss as computed here will
949:     # actually be larger than it should be, by limit * over_limit.sum(), but it
950:     # has the same derivative as the real aux_loss which is penalty * (x.abs() -
951:     # limit).relu().
952:     aux_loss = penalty * ((x_sign * over_limit).to(torch.int8) * x)
953:     # note: we don't do sum() here on aux)_loss, but it's as if we had done
954:     # sum() due to how with_loss() works.
955:     x = with_loss(x, aux_loss, name)
956:     # you must use x for something, or this will be ineffective.
957:     return x

971: def _whitening_metric(x: Tensor, num_groups: int):
972:     """
973:     Computes the "whitening metric", a value which will be 1.0 if all the eigenvalues of
974:     of the centered feature covariance are the same within each group's covariance matrix
975:     and also between groups.
976:     Args:
977:         x: a Tensor of shape (*, num_channels)
978:      num_groups:  the number of groups of channels, a number >=1 that divides num_channels
979:     Returns:
980:         Returns a scalar Tensor that will be 1.0 if the data is "perfectly white" and
981:     greater than 1.0 otherwise.
982:     """
983:     assert x.dtype != torch.float16
984:     x = x.reshape(-1, x.shape[-1])
985:     (num_frames, num_channels) = x.shape
986:     assert num_channels % num_groups == 0
987:     channels_per_group = num_channels // num_groups
988:     x = x.reshape(num_frames, num_groups, channels_per_group).transpose(0, 1)
989:     # x now has shape (num_groups, num_frames, channels_per_group)
990:     # subtract the mean so we use the centered, not uncentered, covariance.
991:     # My experience has been that when we "mess with the gradients" like this,
992:     # it's better not do anything that tries to move the mean around, because
993:     # that can easily cause instability.
994:     x = x - x.mean(dim=1, keepdim=True)
995:     # x_covar: (num_groups, channels_per_group, channels_per_group)
996:     x_covar = torch.matmul(x.transpose(1, 2), x)
997:     x_covar_mean_diag = _diag(x_covar).mean()
998:     # the following expression is what we'd get if we took the matrix product
999:     # of each covariance and measured the mean of its trace, i.e.
1000:     # the same as _diag(torch.matmul(x_covar, x_covar)).mean().
1001:     x_covarsq_mean_diag = (x_covar**2).sum() / (num_groups * channels_per_group)
1002:     # this metric will be >= 1.0; the larger it is, the less 'white' the data was.
1003:     metric = x_covarsq_mean_diag / (x_covar_mean_diag**2 + 1.0e-20)
1004:     return metric

1221: class DoubleSwishFunction(torch.autograd.Function):
1222:     """
1223:       double_swish(x) = x * torch.sigmoid(x-1)
1224: 
1225:     This is a definition, originally motivated by its close numerical
1226:     similarity to swish(swish(x)), where swish(x) =  x * sigmoid(x).
1227: 
1228:     Memory-efficient derivative computation:
1229:      double_swish(x) = x * s, where s(x) = torch.sigmoid(x-1)
1230:      double_swish'(x) = d/dx double_swish(x) =  x * s'(x) + x' * s(x) = x * s'(x) + s(x).
1231:      Now, s'(x) = s(x) * (1-s(x)).
1232:      double_swish'(x) =  x * s'(x) + s(x).
1233:                       =  x * s(x) * (1-s(x)) + s(x).
1234:                      = double_swish(x) * (1-s(x)) + s(x)
1235:      ... so we just need to remember s(x) but not x itself.
1236:     """
1237: 
1238:     @staticmethod
1239:     def forward(ctx, x: Tensor) -> Tensor:
1240:         requires_grad = x.requires_grad
1241:         if x.dtype == torch.float16 or x.dtype == torch.bfloat16:
1242:             x = x.to(torch.float32)
1243: 
1244:         s = torch.sigmoid(x - 1.0)
1245:         y = x * s
1246: 
1247:         if requires_grad:
1248:             deriv = y * (1 - s) + s
1249: 
1250:             # notes on derivative of x * sigmoid(x - 1):
1251:             # https://www.wolframalpha.com/input?i=d%2Fdx+%28x+*+sigmoid%28x-1%29%29
1252:             # min \simeq -0.043638.  Take floor as -0.044 so it's a lower bund
1253:             # max \simeq 1.1990.   Take ceil to be 1.2 so it's an upper bound.
1254:             # the combination of "+ torch.rand_like(deriv)" and casting to torch.uint8 (which
1255:             # floors), should be expectation-preserving.
1256:             floor = -0.044
1257:             ceil = 1.2
1258:             d_scaled = (deriv - floor) * (255.0 / (ceil - floor)) + torch.rand_like(
1259:                 deriv
1260:             )
1261:             if __name__ == "__main__":
1262:                 # for self-testing only.
1263:                 assert d_scaled.min() >= 0.0
1264:                 assert d_scaled.max() < 256.0
1265:             d_int = d_scaled.to(torch.uint8)
1266:             ctx.save_for_backward(d_int)
1267:         if x.dtype == torch.float16 or torch.is_autocast_enabled():
1268:             y = y.to(torch.float16)
1269:         return y
1270: 
1271:     @staticmethod
1272:     def backward(ctx, y_grad: Tensor) -> Tensor:
1273:         (d,) = ctx.saved_tensors
1274:         # the same constants as used in forward pass.
1275:         floor = -0.043637
1276:         ceil = 1.2
1277: 
1278:         d = d * ((ceil - floor) / 255.0) + floor
1279:         return y_grad * d

1515:     @custom_fwd

1542:             "SwooshR": k2.swoosh_r_forward,

1545:         # propagate to the user.
1546:         activation_func = forward_activation_dict[activation]
1547:         x = activation_func(x)
1548:         if dropout_mask is not None:

1551:         return x

1610:              for their backprop (e.g. Balancer or Whiten).

1624:         # create a temporary module of nn.Linear that we'll steal the

1628:         )
